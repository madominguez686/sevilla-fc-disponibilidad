function isZoneManagerRole(rol) { const r = String(rol || "").trim().toLowerCase(); return r === "encargado_zona" || r === "encargado de zona" || r === "admin"; } async function openAppForUser(user) { state.currentUser = user; document.getElementById("loginCard").classList.add("hidden"); document.getElementById("mainApp").classList.remove("hidden"); const esEncargadoZona = isZoneManagerRole(user.rol); const etiquetaRol = esEncargadoZona ? "Encargado de zona" : (user.rol === "encargado" || user.rol === "encargados") ? "Encargado" : "Camarero"; document.getElementById("sessionUserTitle").textContent = user.nombre; document.getElementById("sessionUserMeta").textContent = `Ficha #${user.ficha} · ${user.zona} · ${etiquetaRol}`; const roleTabs = document.getElementById("roleTabs"); await loadExistingAvailabilityForUser(user); if (esEncargadoZona) { // El encargado de zona solo consulta disponibilidades: ocultamos pestañas y su ficha personal roleTabs.classList.add("hidden"); document.getElementById("viewEmp").classList.add("hidden"); document.getElementById("viewEnc").classList.remove("hidden"); renderEncargadoList(); } else { roleTabs.classList.add("hidden"); document.getElementById("viewEnc").classList.add("hidden"); document.getElementById("viewEmp").classList.remove("hidden"); loadEmployeeFicha(user); } } async function switchTab(tab) { const esEncargadoZona = isZoneManagerRole(state.currentUser?.rol); if (esEncargadoZona) { document.getElementById("roleTabs").classList.add("hidden"); document.getElementById("viewEmp").classList.add("hidden"); document.getElementById("viewEnc").classList.remove("hidden"); await loadAllAvailabilitiesForManager(); renderEncargadoList(); return; } const isEmp = tab === "emp"; document.getElementById("tabEmp").classList.toggle("active", isEmp); document.getElementById("tabEnc").classList.toggle("active", !isEmp); document.getElementById("viewEmp").classList.toggle("hidden", !isEmp); document.getElementById("viewEnc").classList.toggle("hidden", isEmp); if (!isEmp) { if (state.currentUser && (state.currentUser.rol === "encargado" || state.currentUser.rol === "admin")) { await loadAllAvailabilitiesForManager(); } renderEncargadoList(); } }
